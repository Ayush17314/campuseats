# Illegal transition (repeat cancellation) — first call 200, second 409
curl -i -X POST http://localhost:5000/orders/ORD-000001/cancellation -H 'Authorization: Bearer demo-token'
HTTP/1.1 200 OK
{"orderId":"ORD-000001","status":"cancelled"}
curl -i -X POST http://localhost:5000/orders/ORD-000001/cancellation -H 'Authorization: Bearer demo-token'
HTTP/1.1 409 CONFLICT
Content-Type: application/problem+json; charset=utf-8
{"type":"https://campuseats.example/problems/illegal-transition","title":"Illegal Transition","status":409,"detail":"A cancelled order cannot be cancelled again."}
